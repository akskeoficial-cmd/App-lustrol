Script SQL para Supabase

Estructura completa: notas_credito, despachos_camion, devoluciones, control_mercaderia, catalogo_barras, historial_repartos, etiquetas_patentes y asignacion_patentes, con índices y políticas RLS.

1. Abrí tu proyecto en Supabase 2. Ingresá al SQL Editor 3. Pegá el script y ejecutá RUN 
CREATE EXTENSION IF NOT EXISTS "uuid-ossp";

-- Notas de Crédito con múltiples motivos
CREATE TABLE IF NOT EXISTS notas_credito (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    codigo_cliente VARCHAR(50) NOT NULL,
    cliente_documento VARCHAR(255) NOT NULL,
    fecha_emision DATE NOT NULL DEFAULT CURRENT_DATE,
    motivos TEXT[] NOT NULL,
    observaciones TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- Tabla de Despachos de Camiones
CREATE TABLE IF NOT EXISTS despachos_camion (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    patente VARCHAR(20) NOT NULL,
    chofer VARCHAR(150),
    fecha DATE NOT NULL DEFAULT CURRENT_DATE,
    cantidad_pedidos INT NOT NULL CHECK (cantidad_pedidos >= 0),
    observaciones TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- Tabla de Devoluciones (Simplificada)
CREATE TABLE IF NOT EXISTS devoluciones (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    acompanante VARCHAR(150) NOT NULL,
    codigo_cliente VARCHAR(50) NOT NULL,
    fecha_emision DATE NOT NULL DEFAULT CURRENT_DATE,
    motivos TEXT[] NOT NULL,
    observaciones TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

CREATE INDEX IF NOT EXISTS idx_dev_fecha_acomp ON devoluciones(fecha_emision, acompanante);

-- Control de Mercadería por Reparto y Planilla/Hoja
CREATE TABLE IF NOT EXISTS control_mercaderia (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    reparto_numero INT NOT NULL,
    hoja_numero INT NOT NULL DEFAULT 1,
    planilla_id VARCHAR(100),
    observacion_origen VARCHAR(100),
    codigo_articulo VARCHAR(100) NOT NULL,
    nombre_articulo VARCHAR(255) NOT NULL,
    ubicacion VARCHAR(50) NOT NULL,
    cantidad_esperada INT NOT NULL,
    cantidad_contada INT DEFAULT 0,
    observacion_control TEXT,
    estado_control VARCHAR(50) DEFAULT 'pendiente',
    fecha_carga DATE NOT NULL DEFAULT CURRENT_DATE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

CREATE INDEX IF NOT EXISTS idx_mercaderia_reparto_hoja ON control_mercaderia(reparto_numero, hoja_numero, estado_control);

-- Catálogo / Diccionario de Códigos de Barras (lector láser)
CREATE TABLE IF NOT EXISTS catalogo_barras (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    codigo_articulo TEXT NOT NULL,
    codigo_barras TEXT NOT NULL,
    descripcion TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

CREATE INDEX IF NOT EXISTS idx_barras_codigo_articulo ON catalogo_barras(codigo_articulo);
CREATE UNIQUE INDEX IF NOT EXISTS idx_barras_codigo_barras ON catalogo_barras(codigo_barras);

-- Historial / Auditoría de Repartos finalizados
CREATE TABLE IF NOT EXISTS historial_repartos (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    reparto_numero INT NOT NULL,
    fecha_auditoria DATE NOT NULL DEFAULT CURRENT_DATE,
    origen VARCHAR(100),
    total_articulos INT NOT NULL DEFAULT 0,
    total_controlados INT NOT NULL DEFAULT 0,
    total_pendientes INT NOT NULL DEFAULT 0,
    total_diferencias INT NOT NULL DEFAULT 0,
    total_observados INT NOT NULL DEFAULT 0,
    unidades_totales INT NOT NULL DEFAULT 0,
    resumen_observaciones TEXT,
    detalle JSONB NOT NULL,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

CREATE INDEX IF NOT EXISTS idx_historial_fecha_reparto ON historial_repartos(fecha_auditoria, reparto_numero);

-- Etiquetas personalizables del módulo "Patentes y Choferes"
-- ⚠️ categoria y texto son NOT NULL: el frontend siempre envía ambos campos.
CREATE TABLE IF NOT EXISTS etiquetas_patentes (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    categoria VARCHAR(30) NOT NULL,  -- 'patente' | 'reparto' | 'zona' | 'chofer' | 'acompanante'
    texto VARCHAR(150) NOT NULL,
    orden INT DEFAULT 0,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

CREATE INDEX IF NOT EXISTS idx_etiquetas_cat ON etiquetas_patentes(categoria, orden);

-- Asignación diaria de patentes (Hoja de Ruta)
CREATE TABLE IF NOT EXISTS asignacion_patentes (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    fecha DATE NOT NULL DEFAULT CURRENT_DATE,
    camion_patente VARCHAR(50) NOT NULL,
    reparto VARCHAR(50),
    zona VARCHAR(100),
    chofer VARCHAR(150),
    acompanante VARCHAR(150),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

CREATE INDEX IF NOT EXISTS idx_asignacion_fecha ON asignacion_patentes(fecha);

-- ✅ Reparación de historial_repartos (por si la tabla se creó con menos columnas)
ALTER TABLE historial_repartos ADD COLUMN IF NOT EXISTS total_pendientes INT NOT NULL DEFAULT 0;
ALTER TABLE historial_repartos ADD COLUMN IF NOT EXISTS total_observados INT NOT NULL DEFAULT 0;
ALTER TABLE historial_repartos ADD COLUMN IF NOT EXISTS unidades_totales INT NOT NULL DEFAULT 0;
ALTER TABLE historial_repartos ADD COLUMN IF NOT EXISTS resumen_observaciones TEXT;

-- 📅 Módulo Ausencias
CREATE TABLE IF NOT EXISTS etiquetas_ausencias (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    categoria VARCHAR(30) NOT NULL,  -- 'persona' | 'motivo'
    texto VARCHAR(150) NOT NULL,
    orden INT DEFAULT 0,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);
-- Si la tabla ya existía con otras columnas, se completan las que faltan
ALTER TABLE etiquetas_ausencias ADD COLUMN IF NOT EXISTS categoria VARCHAR(30) NOT NULL DEFAULT 'persona';
ALTER TABLE etiquetas_ausencias ADD COLUMN IF NOT EXISTS texto VARCHAR(150) NOT NULL DEFAULT '';
ALTER TABLE etiquetas_ausencias ADD COLUMN IF NOT EXISTS orden INT DEFAULT 0;
ALTER TABLE etiquetas_ausencias ADD COLUMN IF NOT EXISTS created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW();
CREATE INDEX IF NOT EXISTS idx_etiq_aus_cat ON etiquetas_ausencias(categoria, orden);

CREATE TABLE IF NOT EXISTS ausencias (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    fecha DATE NOT NULL DEFAULT CURRENT_DATE,
    persona VARCHAR(150) NOT NULL,
    motivo VARCHAR(150) NOT NULL,
    observacion TEXT,
    comprobante_path TEXT,
    comprobante_nombre TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);
-- Si la tabla ya existía con otras columnas, se completan las que faltan
ALTER TABLE ausencias ADD COLUMN IF NOT EXISTS fecha DATE NOT NULL DEFAULT CURRENT_DATE;
ALTER TABLE ausencias ADD COLUMN IF NOT EXISTS persona VARCHAR(150) NOT NULL DEFAULT '';
ALTER TABLE ausencias ADD COLUMN IF NOT EXISTS motivo VARCHAR(150) NOT NULL DEFAULT '';
ALTER TABLE ausencias ADD COLUMN IF NOT EXISTS observacion TEXT;
ALTER TABLE ausencias ADD COLUMN IF NOT EXISTS comprobante_path TEXT;
ALTER TABLE ausencias ADD COLUMN IF NOT EXISTS comprobante_nombre TEXT;
ALTER TABLE ausencias ADD COLUMN IF NOT EXISTS created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW();
CREATE INDEX IF NOT EXISTS idx_ausencias_fecha ON ausencias(fecha);

ALTER TABLE etiquetas_ausencias ENABLE ROW LEVEL SECURITY;
ALTER TABLE ausencias           ENABLE ROW LEVEL SECURITY;
DROP POLICY IF EXISTS "acceso_total_etiq_aus" ON etiquetas_ausencias;
CREATE POLICY "acceso_total_etiq_aus" ON etiquetas_ausencias FOR ALL TO anon USING (true) WITH CHECK (true);
DROP POLICY IF EXISTS "acceso_total_ausencias" ON ausencias;
CREATE POLICY "acceso_total_ausencias" ON ausencias FOR ALL TO anon USING (true) WITH CHECK (true);

-- Bucket de comprobantes (público para poder verlos / descargarlos)
INSERT INTO storage.buckets (id, name, public) VALUES ('comprobantes_ausencias', 'comprobantes_ausencias', true) ON CONFLICT (id) DO NOTHING;
DROP POLICY IF EXISTS "comprobantes_ausencias_all" ON storage.objects;
CREATE POLICY "comprobantes_ausencias_all" ON storage.objects FOR ALL TO anon USING (bucket_id = 'comprobantes_ausencias') WITH CHECK (bucket_id = 'comprobantes_ausencias');

-- Índices adicionales
CREATE INDEX IF NOT EXISTS idx_nc_fecha ON notas_credito(fecha_emision);
CREATE INDEX IF NOT EXISTS idx_nc_motivos_gin ON notas_credito USING GIN(motivos);
CREATE INDEX IF NOT EXISTS idx_despachos_fecha_patente ON despachos_camion(fecha, patente);

-- Políticas RLS de acceso público (publishable key, sin login)
