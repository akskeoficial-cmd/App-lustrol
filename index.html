/* ================================================================
   REPARACIÓN COMPROBANTES v25 — EJECUTAR ESTO PRIMERO
   (idempotente: se puede ejecutar varias veces sin problema)
   Garantiza que los comprobantes de Ausencias y los archivos de
   Documentación guarden la URL COMPLETA del bucket de Supabase
   y que existan las columnas de respaldo. NO toca ni borra datos.
   ================================================================ */

ALTER TABLE ausencias               ADD COLUMN IF NOT EXISTS comprobante_url  TEXT;
ALTER TABLE ausencias               ADD COLUMN IF NOT EXISTS comprobante_path TEXT;
ALTER TABLE ausencias               ADD COLUMN IF NOT EXISTS comprobante_mime VARCHAR(100);
ALTER TABLE ausencias               ADD COLUMN IF NOT EXISTS comprobante_data TEXT;
ALTER TABLE documentacion_registros ADD COLUMN IF NOT EXISTS archivo_url   TEXT;
ALTER TABLE documentacion_registros ADD COLUMN IF NOT EXISTS archivo_mime  VARCHAR(100);
ALTER TABLE documentacion_registros ADD COLUMN IF NOT EXISTS archivo_data  TEXT;

-- Buckets públicos (si no existen aún)
INSERT INTO storage.buckets (id, name, public)
VALUES ('comprobantes_ausencias', 'comprobantes_ausencias', true)
ON CONFLICT (id) DO NOTHING;
INSERT INTO storage.buckets (id, name, public)
VALUES ('documentacion_archivos', 'documentacion_archivos', true)
ON CONFLICT (id) DO NOTHING;

-- Políticas públicas de los buckets (se crean solo si faltan)
DO $pol_comp_v25$
BEGIN
  IF NOT EXISTS (SELECT 1 FROM pg_policies WHERE tablename='objects' AND policyname='lectura_publica_comprobantes') THEN
    CREATE POLICY "lectura_publica_comprobantes" ON storage.objects FOR SELECT TO anon USING (bucket_id = 'comprobantes_ausencias');
  END IF;
  IF NOT EXISTS (SELECT 1 FROM pg_policies WHERE tablename='objects' AND policyname='escritura_publica_comprobantes') THEN
    CREATE POLICY "escritura_publica_comprobantes" ON storage.objects FOR INSERT TO anon WITH CHECK (bucket_id = 'comprobantes_ausencias');
  END IF;
  IF NOT EXISTS (SELECT 1 FROM pg_policies WHERE tablename='objects' AND policyname='borrado_publico_comprobantes') THEN
    CREATE POLICY "borrado_publico_comprobantes" ON storage.objects FOR DELETE TO anon USING (bucket_id = 'comprobantes_ausencias');
  END IF;
  IF NOT EXISTS (SELECT 1 FROM pg_policies WHERE tablename='objects' AND policyname='lectura_publica_documentacion') THEN
    CREATE POLICY "lectura_publica_documentacion" ON storage.objects FOR SELECT TO anon USING (bucket_id = 'documentacion_archivos');
  END IF;
  IF NOT EXISTS (SELECT 1 FROM pg_policies WHERE tablename='objects' AND policyname='escritura_publica_documentacion') THEN
    CREATE POLICY "escritura_publica_documentacion" ON storage.objects FOR INSERT TO anon WITH CHECK (bucket_id = 'documentacion_archivos');
  END IF;
  IF NOT EXISTS (SELECT 1 FROM pg_policies WHERE tablename='objects' AND policyname='borrado_publico_documentacion') THEN
    CREATE POLICY "borrado_publico_documentacion" ON storage.objects FOR DELETE TO anon USING (bucket_id = 'documentacion_archivos');
  END IF;
END
$pol_comp_v25$;

NOTIFY pgrst, 'reload schema';

/* ================================================================
   REPARACIÓN v20.1 — EJECUTAR ESTO PRIMERO
   (Es idempotente: se puede ejecutar varias veces sin problema)

-- 2b) Compatibilizar la tabla ausencias si fue creada con otros nombres
--     (ej: "nombre" en vez de "persona", o sin columnas de comprobante)
ALTER TABLE ausencias ADD COLUMN IF NOT EXISTS persona VARCHAR(200);
ALTER TABLE ausencias ADD COLUMN IF NOT EXISTS comprobante_mime VARCHAR(100);
ALTER TABLE ausencias ADD COLUMN IF NOT EXISTS comprobante_data TEXT;
DO $compat$
BEGIN
  IF EXISTS (SELECT 1 FROM information_schema.columns
             WHERE table_schema='public' AND table_name='ausencias' AND column_name='nombre') THEN

     WHERE (persona IS NULL OR persona = '') AND nombre IS NOT NULL;
    ALTER TABLE ausencias ALTER COLUMN nombre DROP NOT NULL;
  END IF;
END
$compat$;

DECLARE t TEXT;
BEGIN
  FOREACH t IN ARRAY ARRAY['etiquetas_patentes','ausencias','despachos_camion'] LOOP
    IF NOT EXISTS (
      SELECT 1 FROM pg_policies

    END IF;
