Script SQL para Supabase

Estructura completa + REPARACIÓN v20.1 (soluciona "no se guarda").

1. Abrí tu proyecto en Supabase2. Ingresá al SQL Editor3. Pegá el script y ejecutá RUN
/* ================================================================
   REPARACIÓN v20.1 — EJECUTAR ESTO PRIMERO
   (Es idempotente: se puede ejecutar varias veces sin problema)
   Arregla: tabla ausencias, columnas nuevas (acompañante y
   fechas desde/hasta), permisos de escritura y recarga el
   esquema de la API. NO toca ni borra tus datos.
   ================================================================ */

CREATE EXTENSION IF NOT EXISTS "uuid-ossp";

-- 1) Tabla de ausencias (si no existe, se crea COMPLETA)
CREATE TABLE IF NOT EXISTS ausencias (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    persona VARCHAR(200) NOT NULL,
    motivo VARCHAR(100) NOT NULL,
    observacion TEXT,
    fecha DATE NOT NULL DEFAULT CURRENT_DATE,
    fecha_desde DATE,
    fecha_hasta DATE,
    comprobante_url TEXT,
    comprobante_path TEXT,
    comprobante_mime VARCHAR(100),
    comprobante_data TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 2) Columnas nuevas (se agregan solo si faltan)
ALTER TABLE ausencias        ADD COLUMN IF NOT EXISTS fecha_desde DATE;
ALTER TABLE ausencias        ADD COLUMN IF NOT EXISTS fecha_hasta DATE;
ALTER TABLE despachos_camion ADD COLUMN IF NOT EXISTS acompanante VARCHAR(150);

-- 2b) Compatibilizar la tabla ausencias si fue creada con otros nombres
--     (ej: "nombre" en vez de "persona", o sin columnas de comprobante)
ALTER TABLE ausencias ADD COLUMN IF NOT EXISTS persona VARCHAR(200);
ALTER TABLE ausencias ADD COLUMN IF NOT EXISTS comprobante_mime VARCHAR(100);
ALTER TABLE ausencias ADD COLUMN IF NOT EXISTS comprobante_data TEXT;
DO $compat$
BEGIN
  IF EXISTS (SELECT 1 FROM information_schema.columns
             WHERE table_schema='public' AND table_name='ausencias' AND column_name='nombre') THEN
    UPDATE ausencias SET persona = COALESCE(NULLIF(persona,''), nombre)
     WHERE (persona IS NULL OR persona = '') AND nombre IS NOT NULL;
    ALTER TABLE ausencias ALTER COLUMN nombre DROP NOT NULL;
  END IF;
END
$compat$;

-- 2c) Completar asignacion_patentes (hoja de ruta) si le faltan columnas
ALTER TABLE asignacion_patentes ADD COLUMN IF NOT EXISTS camion_patente VARCHAR(50);
ALTER TABLE asignacion_patentes ADD COLUMN IF NOT EXISTS reparto VARCHAR(50);
ALTER TABLE asignacion_patentes ADD COLUMN IF NOT EXISTS zona VARCHAR(255);
ALTER TABLE asignacion_patentes ADD COLUMN IF NOT EXISTS chofer VARCHAR(150);
ALTER TABLE asignacion_patentes ADD COLUMN IF NOT EXISTS acompanante VARCHAR(255);
CREATE INDEX IF NOT EXISTS idx_asignacion_fecha ON asignacion_patentes(fecha);

-- 3) Rellenar el rango de las ausencias anteriores
UPDATE ausencias SET fecha_desde = fecha, fecha_hasta = fecha WHERE fecha_desde IS NULL;

-- 4) Permisos de escritura: RLS + política de acceso total para la app
ALTER TABLE etiquetas_patentes ENABLE ROW LEVEL SECURITY;
ALTER TABLE ausencias          ENABLE ROW LEVEL SECURITY;
ALTER TABLE despachos_camion   ENABLE ROW LEVEL SECURITY;

DO $reparacion$
DECLARE t TEXT;
BEGIN
  FOREACH t IN ARRAY ARRAY['etiquetas_patentes','ausencias','despachos_camion'] LOOP
    IF NOT EXISTS (
      SELECT 1 FROM pg_policies
      WHERE schemaname = 'public' AND tablename = t AND policyname = 'acceso_total_anon'
    ) THEN
      EXECUTE format('CREATE POLICY acceso_total_anon ON %I FOR ALL TO anon USING (true) WITH CHECK (true)', t);
    END IF;
  END LOOP;
END
$reparacion$;

-- 5) Índices
CREATE INDEX IF NOT EXISTS idx_ausencias_rango ON ausencias(fecha_desde, fecha_hasta);
CREATE INDEX IF NOT EXISTS idx_despachos_acomp ON despachos_camion(fecha, acompanante);
CREATE INDEX IF NOT EXISTS idx_etiquetas_cat   ON etiquetas_patentes(categoria, orden);

-- 6) Limpieza de filas de diagnóstico (por si quedó alguna)
DELETE FROM etiquetas_patentes WHERE texto LIKE '_diagnostico%';
DELETE FROM ausencias WHERE persona = '_diagnostico';

-- 7) CLAVE: recarga el esquema de la API de Supabase.
--    Si agregaste columnas y la app "no las ve", con esto se soluciona.
NOTIFY pgrst, 'reload schema';

/* ================================================================
   A CONTINUACIÓN: estructura completa (por si hay que crear
   la base desde cero en otro proyecto). En una base ya creada
   no hace falta ejecutarla, con la REPARACIÓN de arriba alcanza.
   ================================================================ */

CREATE EXTENSION IF NOT EXISTS "uuid-ossp";

CREATE TABLE IF NOT EXISTS notas_credito (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    codigo_cliente VARCHAR(50) NOT NULL,
    cliente_documento VARCHAR(255) NOT NULL,
    fecha_emision DATE NOT NULL DEFAULT CURRENT_DATE,
    motivos TEXT[] NOT NULL,
    observaciones TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS despachos_camion (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    patente VARCHAR(20) NOT NULL,
    chofer VARCHAR(150),
    acompanante VARCHAR(150),
    fecha DATE NOT NULL DEFAULT CURRENT_DATE,
    cantidad_pedidos INT NOT NULL CHECK (cantidad_pedidos >= 0),
    observaciones TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS devoluciones (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    acompanante VARCHAR(150) NOT NULL,
    codigo_cliente VARCHAR(50) NOT NULL,
    fecha_emision DATE NOT NULL DEFAULT CURRENT_DATE,
    motivos TEXT[] NOT NULL,
    observaciones TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS control_mercaderia (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    reparto_numero INT NOT NULL,
    hoja_numero INT NOT NULL DEFAULT 1,
    planilla_id VARCHAR(100),
    observacion_origen VARCHAR(100),
    codigo_articulo VARCHAR(100) NOT NULL,
    nombre_articulo VARCHAR(255) NOT NULL,
    ubicacion VARCHAR(50) NOT NULL,
    cantidad_esperada INT NOT NULL,
    cantidad_contada INT DEFAULT 0,
    observacion_control TEXT,
    estado_control VARCHAR(50) DEFAULT 'pendiente',
    fecha_carga DATE NOT NULL DEFAULT CURRENT_DATE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS catalogo_barras (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    codigo_articulo TEXT NOT NULL,
    codigo_barras TEXT NOT NULL,
    descripcion TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS historial_repartos (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    reparto_numero INT NOT NULL,
    fecha_auditoria DATE NOT NULL DEFAULT CURRENT_DATE,
    origen VARCHAR(100),
    total_articulos INT NOT NULL DEFAULT 0,
    total_controlados INT NOT NULL DEFAULT 0,
    total_pendientes INT NOT NULL DEFAULT 0,
    total_diferencias INT NOT NULL DEFAULT 0,
    unidades_totales INT NOT NULL DEFAULT 0,
    resumen_observaciones TEXT,
    detalle JSONB NOT NULL,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS etiquetas_patentes (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    categoria VARCHAR(30) NOT NULL,
    texto VARCHAR(150) NOT NULL,
    orden INT DEFAULT 0,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS asignacion_patentes (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    fecha DATE NOT NULL DEFAULT CURRENT_DATE,
    camion_patente VARCHAR(50) NOT NULL,
    reparto VARCHAR(50),
    zona VARCHAR(255),
    chofer VARCHAR(150),
    acompanante VARCHAR(255),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- Tabla de Ausencias (comprobante en bucket o embebido como fallback)
CREATE TABLE IF NOT EXISTS ausencias (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    persona VARCHAR(200) NOT NULL,
    motivo VARCHAR(100) NOT NULL,
    observacion TEXT,
    fecha DATE NOT NULL DEFAULT CURRENT_DATE,
    fecha_desde DATE,
    fecha_hasta DATE,
    comprobante_url TEXT,
    comprobante_path TEXT,
    comprobante_mime VARCHAR(100),
    comprobante_data TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- Bucket público para comprobantes de ausencias
INSERT INTO storage.buckets (id, name, public)
VALUES ('comprobantes_ausencias', 'comprobantes_ausencias', true)
ON CONFLICT (id) DO NOTHING;

CREATE INDEX IF NOT EXISTS idx_dev_fecha_acomp ON devoluciones(fecha_emision, acompanante);
CREATE INDEX IF NOT EXISTS idx_mercaderia_reparto_hoja ON control_mercaderia(reparto_numero, hoja_numero, estado_control);
CREATE INDEX IF NOT EXISTS idx_barras_codigo_articulo ON catalogo_barras(codigo_articulo);
CREATE UNIQUE INDEX IF NOT EXISTS idx_barras_codigo_barras ON catalogo_barras(codigo_barras);
CREATE INDEX IF NOT EXISTS idx_historial_fecha_reparto ON historial_repartos(fecha_auditoria, reparto_numero);
CREATE INDEX IF NOT EXISTS idx_etiquetas_cat ON etiquetas_patentes(categoria, orden);
CREATE INDEX IF NOT EXISTS idx_asignacion_fecha ON asignacion_patentes(fecha);
CREATE INDEX IF NOT EXISTS idx_ausencias_fecha ON ausencias(fecha);
CREATE INDEX IF NOT EXISTS idx_ausencias_persona ON ausencias(persona);
CREATE INDEX IF NOT EXISTS idx_nc_fecha ON notas_credito(fecha_emision);
CREATE INDEX IF NOT EXISTS idx_nc_motivos_gin ON notas_credito USING GIN(motivos);
CREATE INDEX IF NOT EXISTS idx_despachos_fecha_patente ON despachos_camion(fecha, patente);

ALTER TABLE notas_credito      ENABLE ROW LEVEL SECURITY;
ALTER TABLE despachos_camion   ENABLE ROW LEVEL SECURITY;
ALTER TABLE devoluciones       ENABLE ROW LEVEL SECURITY;
ALTER TABLE control_mercaderia ENABLE ROW LEVEL SECURITY;
ALTER TABLE catalogo_barras    ENABLE ROW LEVEL SECURITY;
ALTER TABLE historial_repartos ENABLE ROW LEVEL SECURITY;
ALTER TABLE etiquetas_patentes ENABLE ROW LEVEL SECURITY;
ALTER TABLE asignacion_patentes ENABLE ROW LEVEL SECURITY;
ALTER TABLE ausencias          ENABLE ROW LEVEL SECURITY;

CREATE POLICY "acceso_total_notas"      ON notas_credito      FOR ALL TO anon USING (true) WITH CHECK (true);
CREATE POLICY "acceso_total_despachos"  ON despachos_camion   FOR ALL TO anon USING (true) WITH CHECK (true);
CREATE POLICY "acceso_total_devol"      ON devoluciones       FOR ALL TO anon USING (true) WITH CHECK (true);
CREATE POLICY "acceso_total_mercaderia" ON control_mercaderia FOR ALL TO anon USING (true) WITH CHECK (true);
CREATE POLICY "acceso_total_barras"     ON catalogo_barras    FOR ALL TO anon USING (true) WITH CHECK (true);
CREATE POLICY "acceso_total_historial"  ON historial_repartos FOR ALL TO anon USING (true) WITH CHECK (true);
CREATE POLICY "acceso_total_etiquetas"  ON etiquetas_patentes FOR ALL TO anon USING (true) WITH CHECK (true);
CREATE POLICY "acceso_total_asignacion" ON asignacion_patentes FOR ALL TO anon USING (true) WITH CHECK (true);
CREATE POLICY "acceso_total_ausencias"  ON ausencias          FOR ALL TO anon USING (true) WITH CHECK (true);

-- Políticas del bucket de comprobantes
CREATE POLICY "lectura_publica_comprobantes"  ON storage.objects FOR SELECT TO anon USING (bucket_id = 'comprobantes_ausencias');
CREATE POLICY "escritura_publica_comprobantes" ON storage.objects FOR INSERT TO anon WITH CHECK (bucket_id = 'comprobantes_ausencias');
CREATE POLICY "borrado_publico_comprobantes"   ON storage.objects FOR DELETE TO anon USING (bucket_id = 'comprobantes_ausencias');

-- ===== MIGRACIÓN v20 (ejecutar también en bases ya creadas · es idempotente) =====
ALTER TABLE despachos_camion ADD COLUMN IF NOT EXISTS acompanante VARCHAR(150);
ALTER TABLE ausencias ADD COLUMN IF NOT EXISTS fecha_desde DATE;
ALTER TABLE ausencias ADD COLUMN IF NOT EXISTS fecha_hasta DATE;
UPDATE ausencias SET fecha_desde = fecha, fecha_hasta = fecha WHERE fecha_desde IS NULL;
CREATE INDEX IF NOT EXISTS idx_despachos_acomp ON despachos_camion(fecha, acompanante);
CREATE INDEX IF NOT EXISTS idx_ausencias_rango ON ausencias(fecha_desde, fecha_hasta);
NOTIFY pgrst, 'reload schema';
